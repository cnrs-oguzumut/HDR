\begin{frame}{Modeling: Deformation Gradient and Strain Tensor}
    
    % --- Custom TikZ Infographic for Deformation ---
    \begin{center}
        \begin{tikzpicture}[>=stealth, scale=0.8, every node/.style={transform shape}]
            % Reference configuration
            \draw[fill=AcademicNavy!5, draw=AcademicNavy, thick] (0,0) rectangle (2,2);
            \node at (1,1) {$\mathbf{X}$};
            \node[font=\footnotesize, AcademicNavy] at (1,-0.4) {$\mathbf{I}$};
            
            % Mapping Arrow
            \draw[->, thick, AcademicNavy] (2.8,1) -- (4.2,1);
            
            % Deformed configuration
            \begin{scope}[shift={(5.0,0.2)}]
                \draw[fill=SecondaryTitle!5, draw=SecondaryTitle, thick] plot [smooth cycle] coordinates {(0,0.5) (1,0) (2.5,0.5) (2.8,1.5) (1.5,2.2) (0.2,1.5)};
                \node at (1.4,1.1) {$\mathbf{Y}(\mathbf{X})$};
                \node[font=\footnotesize, SecondaryTitle] at (1.4,-0.6) {$\mathbf{F}$};
            \end{scope}
        \end{tikzpicture}
    \end{center}
    
    \vspace{-0.2cm}
    
    % --- Equations and Text ---
    \small
    \begin{itemize}
        \item \textbf{\color{AcademicNavy} Deformed state:}
        \begin{itemize}
            \item Displacement field: \hspace{0.5cm} $\mathbf{u}(\mathbf{X}) = \mathbf{Y}(\mathbf{X}) - \mathbf{X}$
            \item Deformation gradient: \hspace{0.25cm} $\mathbf{F} = \nabla \mathbf{Y}$
        \end{itemize}
        
        \vspace{0.2cm}
        \item \textbf{\color{AcademicNavy} Strain energy defined in terms of strain tensor:}
        \vspace{-0.1cm}
        \begin{itemize}
            \item Geometrically nonlinear strain tensor:
            \vspace{-0.15cm}
            \begin{equation*}
                \epsilon_{ij}^{NL} = \frac{1}{2}(\mathbf{F}^T \mathbf{F} - \mathbf{I}) = \frac{1}{2} \left( \frac{\partial u_i}{\partial x_j} + \frac{\partial u_j}{\partial x_i} + \sum_k \frac{\partial u_k}{\partial x_i} \frac{\partial u_k}{\partial x_j} \right)
            \end{equation*}
            \vspace{-0.3cm}
            \item Geometrically linear strain tensor:
            \vspace{-0.15cm}
            \begin{equation*}
                \epsilon_{ij}^{L} = \frac{1}{2}(\mathbf{F}^T + \mathbf{F} - 2\mathbf{I}) = \frac{1}{2} \left( \frac{\partial u_i}{\partial x_j} + \frac{\partial u_j}{\partial x_i} \right)
            \end{equation*}
        \end{itemize}
    \end{itemize}
\end{frame}

    \vspace{0.4cm}
    
    % --- Equations and Text ---
    \begin{itemize}
        \item nonlinear: \hspace{0.5cm} $\displaystyle \epsilon_{ij}^{NL} = \frac{1}{2} \left[ (\mathbf{R}\mathbf{F})^T (\mathbf{R}\mathbf{F}) - \mathbf{I} \right] = \frac{1}{2} (\mathbf{F}^T \mathbf{F} - \mathbf{I})$
        
        \vspace{0.3cm}
        \item linear: \hspace{1.07cm} $\displaystyle \epsilon_{ij}^{L}\; = \frac{1}{2} \left[ (\mathbf{R}\mathbf{F})^T + (\mathbf{R}\mathbf{F}) - 2\mathbf{I} \right] \neq \frac{1}{2} (\mathbf{F}^T + \mathbf{F} - 2\mathbf{I})$
        
        \vspace{0.3cm}
        \item Strain energy defined in terms of $\epsilon_{ij}^{NL}$ or $\epsilon_{ij}^{L}$ strain tensor
    \end{itemize}

    \begin{center}
        {\large \color{SecondaryTitle} $\rightarrow$} \textbf{\large In linear modeling, energy is not invariant under rotation!!!}
    \end{center}

\begin{frame}{Lagrange reduction}
    
    \vspace{0.4cm}
    \begin{itemize}
        \setlength\itemsep{0.3cm}
        \item \textbf{if} $C_{12} < 0$, change sign to $C_{12}$;
        \item \textbf{if} $C_{22} < C_{11}$, swap the two components;
        \item \textbf{if} $2C_{12} > C_{11}$, set $C_{12} = C_{12} - C_{11}$ and $C_{22} = C_{22} + C_{11} - 2C_{12}$.
    \end{itemize}
    
    \vspace{0.8cm}
    \begin{itemize}
        \setlength\itemsep{0.3cm}
        \item initiate $\mathbf{m} = \mathbb{I}$;
        \item define the following three matrices : \quad
              $\displaystyle \mathbf{m}_1 = \begin{pmatrix} 1 & 0 \\ 0 & -1 \end{pmatrix}, \quad \mathbf{m}_2 = \begin{pmatrix} 0 & 1 \\ 1 & 0 \end{pmatrix}, \quad \mathbf{m}_3 = \begin{pmatrix} 1 & -1 \\ 0 & 1 \end{pmatrix}$;
        \item initiate recursive algorithm :
        \item \textbf{if} $C_{12} < 0$, \quad $\mathbf{m} \rightarrow \mathbf{m}\mathbf{m}_1$;
        \item \textbf{if} $C_{22} < C_{11}$, \quad $\mathbf{m} \rightarrow \mathbf{m}\mathbf{m}_2$;
        \item \textbf{if} $2C_{12} > C_{11}$, \quad $\mathbf{m} \rightarrow \mathbf{m}\mathbf{m}_3$.
    \end{itemize}

        % --- Right: Summary Text ---
        \begin{column}{0.48\textwidth}
            \colorbox{gray!15}{
                \begin{minipage}{\dimexpr\textwidth-12pt\relax}
                    \scriptsize
                    \textbf{Observables:} \\
                    Energy dissipated at a single avalanche $\Delta W$ and the corresponding macroscopic stress drop $\Delta \Sigma = A\Delta\sigma_{xy}$ ($A$: surface area).
                    
                    \vspace{0.3cm}
                    
                    \textbf{Finite-size scaling ansatz:} \\
                    Probability distribution of these observables follows:
                    \begin{equation*}
                        P(s;L) = s^{-x} F(s/L^{D_x})
                    \end{equation*}
                    where $L = Nh$ is linear system size, $h$ is element size, $x$ is power-law exponent ($x = \tau$ for stress, $x = \epsilon$ for energy), and $D_x$ is fractal dimension. $F(u)$ is a universal scaling function.
                    
                    \vspace{0.2cm}
                    
                    \textbf{Key Findings:} \\
                    In amorphous plasticity, exponents $\epsilon$ and $\tau$ are similar for pre- and post-yield regimes, while $D_\epsilon$ and $D_\tau$ may differ [15, 63--65].
                \end{minipage}
            }
        \end{column}
    \end{columns}
\end{frame}

% Slide 35: Extension: Iterative Rebonding Algorithm
\begin{frame}{Extension: Iterative Rebonding Algorithm}
    \begin{columns}[c, onlytextwidth]
        \begin{column}{0.62\textwidth}
            \begin{enumerate}
                \item \textbf{\color{AcademicNavy} Standard Energy Minimization}
                \begin{itemize}
                    \item Perform L-BFGS for $W(\mathbf{u}, \mathbf{M})$ at fixed loading step.
                \end{itemize}
                
                \item \textbf{\color{AcademicNavy} Large-Shear Activation ($m_3$)}
                \begin{itemize}
                    \item Identify nodes/points where the \textbf{3rd step of Lagrange reduction ($m_3$)} is triggered.
                \end{itemize}
                
                \item \textbf{\color{AcademicNavy} Global Relaxation Loop}
                \begin{itemize}
                    \item \textit{While energy $W$ is decreasing:}
                    \begin{itemize}
                        \item \textbf{Re-bond:} Global connectivity update using current positions $\mathbf{x} = \mathbf{X} + \mathbf{u}$.
                        \item \textbf{Minimize:} Unconstrained relaxation of the energy with the new topology.
                    \end{itemize}
                    \item Convergence reached when the energy steady state is found.
                \end{itemize}
            \end{enumerate}
        \end{column}
        
        \begin{column}{0.35\textwidth}
            \centering
            \begin{tikzpicture}[>=stealth, thick, node distance=1.1cm]
                \node[draw, rounded corners, fill=gray!10] (min1) {\footnotesize $\arg\min_{\mathbf{u}} W(\mathbf{u})$};
                \node[draw, diamond, aspect=2, below=of min1, fill=white, inner sep=1pt] (check) {\scriptsize $m_3$ active?};
                \node[draw, rounded corners, below=of check, fill=AcademicNavy!20] (reb) {\footnotesize Re-bond};
                \node[draw, rounded corners, below=of reb, fill=gray!10] (min2) {\footnotesize Full Min.};
                \node[draw, diamond, aspect=2, below=of min2, fill=white, inner sep=1pt] (check2) {\tiny $\Delta W \approx 0$?};
                \node[draw, rounded corners, right=of check, xshift=-0.2cm, fill=AcademicNavy!10] (next) {\scriptsize Success};

                \draw[->] (check) -- node[left] {\tiny Yes} (reb);
                \draw[->] (reb) -- (min2);
                \draw[->] (min2) -- (check2);
                \draw[->] (check2.west) -- ++(-0.5,0) |- (reb.west) node[near start, left] {\tiny No};
                \draw[->] (check2.east) -| (next.south) node[near start, above] {\tiny Yes};
