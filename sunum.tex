% Slide 3
\section{Introduction}
\begin{frame}{Scientific Context: Nonlinearity \& Instability}
    \begin{block}{Scientific Challenges}
        \vspace{0.1cm}
        \textbf{\color{AcademicNavy} The Challenge:}\\[0.1cm]
        Understanding material behavior beyond the elastic limit where microscale rearrangements drive macroscopic nonlinearity.
        
        \vspace{0.3cm}
        \textbf{\color{AcademicNavy} Key Phenomena:}
        \begin{itemize}
            \item \textbf{Phase Transitions:} Martensitic transformations.
            \item \textbf{Plasticity:} Dislocation-mediated flow and avalanches.
            \item \textbf{Fracture:} Brittle-to-ductile transitions and damage.
        \end{itemize}
    \end{block}

    \vfill

    \begin{center}
        \begin{minipage}{0.85\textwidth}
            \begin{block}{Unifying Perspective}
                \centering
                \vspace{0.1cm}
                These phenomena are inherently \textbf{\color{AcademicNavy} displacement-driven} (\textit{non-diffusive}) and can be studied within the \textbf{same context} of displacive instabilities.
            \end{block}
        \end{minipage}
    \end{center}
\end{frame}
